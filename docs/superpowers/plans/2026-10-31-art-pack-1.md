# Art Pack 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Produce Neo Meridian's first professional pixel-art pack (tileset, 4 district building kits, props, 4 vehicle skins) as palette-indexed TypeScript sources exported to PNG atlases plus `atlas.json`, reviewable in a gallery page.

**Architecture:** `art/` is a Node-only authoring toolchain run with Node's built-in TypeScript type stripping. Assets are `PixelCanvas` objects holding palette indices. Families are authored either as text grids (detailed sprites) or drawing functions (repeating textures). A shelf packer writes albedo and emissive atlases. A second Vite entry (`assets.html`) renders the atlases for review.

**Tech Stack:** Node 22 (type stripping), `node:zlib`, Vite 8 multi-page, Vitest 5, Higgsfield MCP (concept art only).

## Global Constraints

- Original designs only. The reference games guide the *style*; nothing is copied or traced.
- 16×16 px per tile = 1 world unit. One facade or roof module is 16×16.
- 48-colour master palette plus index 0 = transparent. Variants are index remaps only.
- Outlines use the `o0..o2` ramp, never pure black. Light comes from the top-left. No anti-aliasing.
- Every asset has emissive frames matching its albedo frames in count and size.
- `art/` code must be Node type-strip compatible: `import type` for types, explicit `.ts` import
  extensions, and no enums, namespaces or constructor parameter properties.
- No new npm dependencies.
- Generated files in `public/assets/` are committed; the build must be deterministic.

## Note on asset content

Tasks 1–3 and 8 give complete code. Tasks 4–7 give complete tests, frame lists, sizes and
authoring rules, but the pixel content itself is creative work authored during execution
against the approved concept art. Each of those tasks has a mandatory visual review step
(gallery screenshot) before commit.

```ts
/** Indexed-colour drawing surface. Values are palette indices; 0 = transparent. */
export class PixelCanvas {
  readonly w: number;
  readonly h: number;
  readonly data: Uint8Array;

  constructor(w: number, h: number, data?: Uint8Array) {
    this.w = w;
    this.h = h;
    this.data = data ?? new Uint8Array(w * h);
  }

  get(x: number, y: number): number {
    if (x < 0 || y < 0 || x >= this.w || y >= this.h) return 0;
    return this.data[y * this.w + x] as number;
  }

  set(x: number, y: number, i: number): void {
    if (x < 0 || y < 0 || x >= this.w || y >= this.h) return;
    this.data[y * this.w + x] = i;
  }

  rect(x: number, y: number, w: number, h: number, i: number): this {
    for (let yy = y; yy < y + h; yy++) for (let xx = x; xx < x + w; xx++) this.set(xx, yy, i);
    return this;
  }

  line(x0: number, y0: number, x1: number, y1: number, i: number): this {
    const dx = Math.abs(x1 - x0), sx = x0 < x1 ? 1 : -1;
    const dy = -Math.abs(y1 - y0), sy = y0 < y1 ? 1 : -1;
    let err = dx + dy;
    for (;;) {
      this.set(x0, y0, i);
      if (x0 === x1 && y0 === y1) break;
      const e2 = 2 * err;
      if (e2 >= dy) { err += dy; x0 += sx; }
      if (e2 <= dx) { err += dx; y0 += sy; }
    }
    return this;
  }

  /** Checkerboard of a/b (a where (x+y) is even). */
  dither(x: number, y: number, w: number, h: number, a: number, b: number): this {
    for (let yy = y; yy < y + h; yy++) for (let xx = x; xx < x + w; xx++) this.set(xx, yy, (xx + yy) % 2 === 0 ? a : b);
    return this;
  }

  blit(src: PixelCanvas, dx: number, dy: number, flipX = false): this {
    for (let y = 0; y < src.h; y++) {
      for (let x = 0; x < src.w; x++) {
        const v = src.get(flipX ? src.w - 1 - x : x, y);
        if (v !== 0) this.set(dx + x, dy + y, v);
      }
    }
    return this;
  }

  /** Paint `i` on every transparent pixel 4-adjacent to an opaque one. */
  outline(i: number): this {
    const src = this.clone();
    for (let y = 0; y < this.h; y++) {
      for (let x = 0; x < this.w; x++) {
        if (src.get(x, y) !== 0) continue;
        if (src.get(x - 1, y) || src.get(x + 1, y) || src.get(x, y - 1) || src.get(x, y + 1)) this.set(x, y, i);
      }
    }
    return this;
  }

  remap(table: Readonly<Record<number, number>>): PixelCanvas {
    const out = this.clone();
    for (let k = 0; k < out.data.length; k++) {
      const v = out.data[k] as number;
      const r = table[v];
      if (r !== undefined) out.data[k] = r;
    }
    return out;
  }

  clone(): PixelCanvas {
    return new PixelCanvas(this.w, this.h, new Uint8Array(this.data));
  }

  flipX(): PixelCanvas {
    return new PixelCanvas(this.w, this.h).blit(this, 0, 0, true);
  }
}
```

```ts
import { PixelCanvas } from './pixels.ts';

/**
 * Parse a text-grid sprite. Common indentation is removed; blank leading/trailing lines ignored.
 * '.' is always transparent; every other character must be in `legend`.
 */
export function parseGrid(src: string, legend: Readonly<Record<string, number>>): PixelCanvas {
  const lines = src.split('\n').map((l) => l.replace(/\s+$/, ''));
  while (lines.length && lines[0] === '') lines.shift();
  while (lines.length && lines[lines.length - 1] === '') lines.pop();
  if (lines.length === 0) throw new Error('parseGrid: empty grid');
  const indent = Math.min(...lines.map((l) => (l.match(/^ */) as RegExpMatchArray)[0].length));
  const rows = lines.map((l) => l.slice(indent));
  const w = (rows[0] as string).length;
  const c = new PixelCanvas(w, rows.length);
  rows.forEach((row, y) => {
    if (row.length !== w) throw new Error(`parseGrid: ragged row ${y} (${row.length} != ${w})`);
    for (let x = 0; x < w; x++) {
      const ch = row[x] as string;
      if (ch === '.') continue;
      const v = legend[ch];
      if (v === undefined) throw new Error(`parseGrid: unknown character '${ch}' at ${x},${y}`);
      c.set(x, y, v);
    }
  });
  return c;
}
```

```ts
import { deflateSync, inflateSync } from 'node:zlib';

const CRC_TABLE = (() => {
  const t = new Uint32Array(256);
  for (let n = 0; n < 256; n++) {
    let c = n;
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
    t[n] = c >>> 0;
  }
  return t;
})();

function crc32(buf: Uint8Array): number {
  let c = 0xffffffff;
  for (const b of buf) c = (CRC_TABLE[(c ^ b) & 0xff] as number) ^ (c >>> 8);
  return (c ^ 0xffffffff) >>> 0;
}

function chunk(type: string, data: Uint8Array): Buffer {
  const len = Buffer.alloc(4);
  len.writeUInt32BE(data.length);
  const td = Buffer.concat([Buffer.from(type, 'ascii'), data]);
  const crc = Buffer.alloc(4);
  crc.writeUInt32BE(crc32(td));
  return Buffer.concat([len, td, crc]);
}

const SIGNATURE = Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]);

/** Encode 8-bit RGBA pixels as PNG (filter 0, deterministic). */
export function encodePng(w: number, h: number, rgba: Uint8Array): Buffer {
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(w, 0);
  ihdr.writeUInt32BE(h, 4);
  ihdr[8] = 8; // bit depth
  ihdr[9] = 6; // colour type RGBA
  const raw = Buffer.alloc(h * (w * 4 + 1));
  for (let y = 0; y < h; y++) {
    raw[y * (w * 4 + 1)] = 0;
    Buffer.from(rgba.buffer, rgba.byteOffset + y * w * 4, w * 4).copy(raw, y * (w * 4 + 1) + 1);
  }
  return Buffer.concat([SIGNATURE, chunk('IHDR', ihdr), chunk('IDAT', deflateSync(raw, { level: 9 })), chunk('IEND', new Uint8Array(0))]);
}

/** Minimal decoder for PNGs produced by encodePng (filter 0 only). Used by tests. */
export function decodePng(buf: Buffer): { w: number; h: number; rgba: Uint8Array } {
  if (!buf.subarray(0, 8).equals(SIGNATURE)) throw new Error('not a PNG');
  let off = 8;
  let w = 0, h = 0;
  const idat: Buffer[] = [];
  while (off < buf.length) {
    const len = buf.readUInt32BE(off);
    const type = buf.toString('ascii', off + 4, off + 8);
    const data = buf.subarray(off + 8, off + 8 + len);
    if (crc32(buf.subarray(off + 4, off + 8 + len)) !== buf.readUInt32BE(off + 8 + len)) throw new Error(`bad CRC in ${type}`);
    if (type === 'IHDR') { w = data.readUInt32BE(0); h = data.readUInt32BE(4); }
    if (type === 'IDAT') idat.push(data);
    off += 12 + len;
  }
  const raw = inflateSync(Buffer.concat(idat));
  const rgba = new Uint8Array(w * h * 4);
  for (let y = 0; y < h; y++) {
    if (raw[y * (w * 4 + 1)] !== 0) throw new Error('unsupported filter');
    rgba.set(raw.subarray(y * (w * 4 + 1) + 1, (y + 1) * (w * 4 + 1)), y * w * 4);
  }
  return { w, h, rgba };
}
```

```ts
import { PixelCanvas } from './pixels.ts';
import type { Rect } from './atlas.ts';

export type AssetDef = {
  name: string;
  frames: PixelCanvas[];
  emissive: PixelCanvas[];
  fps?: number;
  parts?: Record<string, Rect>;
};

/** Emissive mask: keep only pixels whose index is in `glow`. */
export function glowMask(frame: PixelCanvas, glow: readonly number[]): PixelCanvas {
  const set = new Set(glow);
  const out = new PixelCanvas(frame.w, frame.h);
  for (let k = 0; k < frame.data.length; k++) {
    const v = frame.data[k] as number;
    if (set.has(v)) out.data[k] = v;
  }
  return out;
}

export function makeAsset(
  name: string,
  frames: PixelCanvas[],
  glow: readonly number[],
  opts: { fps?: number; parts?: Record<string, Rect> } = {},
): AssetDef {
  return { name, frames, emissive: frames.map((f) => glowMask(f, glow)), ...opts };
}

export function remapAsset(def: AssetDef, table: Readonly<Record<number, number>>, name: string): AssetDef {
  return { ...def, name, frames: def.frames.map((f) => f.remap(table)), emissive: def.emissive.map((f) => f.remap(table)) };
}
```

```ts
import { PixelCanvas } from './pixels.ts';
import { encodePng } from './png.ts';
import { rgbaOf } from '../palette.ts';
import type { AssetDef } from './asset.ts';

export type Rect = { x: number; y: number; w: number; h: number };
export type AtlasMeta = {
  width: number;
  height: number;
  frames: Record<string, Rect[]>;
  fps: Record<string, number>;
  parts: Record<string, Record<string, Rect>>;
};
export type Packed = { albedo: PixelCanvas; emissive: PixelCanvas; entry: AtlasMeta };

const nextPow2 = (n: number) => 2 ** Math.ceil(Math.log2(Math.max(1, n)));
const cmp = (a: string, b: string) => (a < b ? -1 : a > b ? 1 : 0);

/** Deterministic shelf packer: tallest first, then by name, then frame index. */
export function packAtlas(defs: AssetDef[], width = 256, padding = 1): Packed {
  const seen = new Set<string>();
  for (const d of defs) {
    if (seen.has(d.name)) throw new Error(`packAtlas: duplicate asset '${d.name}'`);
    seen.add(d.name);
    if (d.frames.length === 0) throw new Error(`packAtlas: '${d.name}' has no frames`);
    if (d.emissive.length !== d.frames.length) throw new Error(`packAtlas: '${d.name}' emissive frame count mismatch`);
    d.frames.forEach((f, i) => {
      const e = d.emissive[i] as PixelCanvas;
      if (e.w !== f.w || e.h !== f.h) throw new Error(`packAtlas: '${d.name}' emissive size mismatch`);
      if (f.w !== (d.frames[0] as PixelCanvas).w || f.h !== (d.frames[0] as PixelCanvas).h) throw new Error(`packAtlas: '${d.name}' frame sizes differ`);
      if (f.w + 2 * padding > width) throw new Error(`packAtlas: '${d.name}' is too wide for ${width}px atlas`);
    });
  }
  const items = defs.flatMap((d) => d.frames.map((f, i) => ({ d, i, f })));
  const order = [...items].sort((a, b) => b.f.h - a.f.h || cmp(a.d.name, b.d.name) || a.i - b.i);
  const placed = new Map<string, Rect>();
  let x = padding, y = padding, shelf = 0;
  for (const it of order) {
    if (x + it.f.w + padding > width) { x = padding; y += shelf + padding; shelf = 0; }
    placed.set(`${it.d.name}#${it.i}`, { x, y, w: it.f.w, h: it.f.h });
    x += it.f.w + padding;
    shelf = Math.max(shelf, it.f.h);
  }
  const height = nextPow2(y + shelf + padding);
  const albedo = new PixelCanvas(width, height);
  const emissive = new PixelCanvas(width, height);
  const entry: AtlasMeta = { width, height, frames: {}, fps: {}, parts: {} };
  for (const d of defs) {
    entry.frames[d.name] = d.frames.map((f, i) => {
      const r = placed.get(`${d.name}#${i}`) as Rect;
      albedo.blit(f, r.x, r.y);
      emissive.blit(d.emissive[i] as PixelCanvas, r.x, r.y);
      return r;
    });
    if (d.fps) entry.fps[d.name] = d.fps;
    if (d.parts) entry.parts[d.name] = d.parts;
  }
  return { albedo, emissive, entry };
}

export function canvasToPng(c: PixelCanvas): Buffer {
  const rgba = new Uint8Array(c.w * c.h * 4);
  for (let k = 0; k < c.data.length; k++) rgba.set(rgbaOf(c.data[k] as number), k * 4);
  return encodePng(c.w, c.h, rgba);
}
```

```ts
import { mkdirSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { packAtlas, canvasToPng, type AtlasMeta } from './lib/atlas.ts';
import type { AssetDef } from './lib/asset.ts';

export const OUT_DIR = 'public/assets';

/** Every atlas and its assets. Families register here. */
export function atlasSources(): Record<string, AssetDef[]> {
  const out: Record<string, AssetDef[]> = {};
  return out;
}

export function buildAll(): Map<string, Buffer> {
  const files = new Map<string, Buffer>();
  const atlases: Record<string, AtlasMeta & { image: string; emissive: string }> = {};
  for (const [name, defs] of Object.entries(atlasSources())) {
    const packed = packAtlas(defs, 256, 1);
    const image = `${name}.png`;
    const emissive = `${name}_emissive.png`;
    files.set(image, canvasToPng(packed.albedo));
    files.set(emissive, canvasToPng(packed.emissive));
    atlases[name] = { image, emissive, ...packed.entry };
  }
  files.set('atlas.json', Buffer.from(`${JSON.stringify({ tileSize: 16, atlases }, null, 1)}\n`));
  return files;
}
```

```ts
import './gallery.css';

type Rect = { x: number; y: number; w: number; h: number };
type Atlas = { image: string; emissive: string; width: number; height: number; frames: Record<string, Rect[]>; fps: Record<string, number> };
type AtlasFile = { tileSize: number; atlases: Record<string, Atlas> };
type Card = { canvas: HTMLCanvasElement; frames: Rect[]; fps: number; albedo: HTMLImageElement; glow: HTMLImageElement };

const BASE = '/assets/';
const zoomEl = document.getElementById('zoom') as HTMLInputElement;
const nightEl = document.getElementById('night') as HTMLInputElement;
const root = document.getElementById('root') as HTMLElement;
const cards: Card[] = [];

const loadImage = (src: string) =>
  new Promise<HTMLImageElement>((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error(`failed to load ${src}`));
    img.src = src;
  });

function draw(card: Card, t: number): void {
  const zoom = Number(zoomEl.value);
  const frame = card.frames[card.fps ? Math.floor(t * card.fps) % card.frames.length : 0] as Rect;
  const c = card.canvas;
  if (c.width !== frame.w * zoom) { c.width = frame.w * zoom; c.height = frame.h * zoom; }
  const ctx = c.getContext('2d') as CanvasRenderingContext2D;
  ctx.imageSmoothingEnabled = false;
  ctx.globalCompositeOperation = 'source-over';
  ctx.clearRect(0, 0, c.width, c.height);
  ctx.drawImage(card.albedo, frame.x, frame.y, frame.w, frame.h, 0, 0, c.width, c.height);
  if (nightEl.checked) {
    ctx.globalCompositeOperation = 'source-atop';
    ctx.fillStyle = 'rgba(12, 8, 40, 0.62)';
    ctx.fillRect(0, 0, c.width, c.height);
    ctx.globalCompositeOperation = 'lighter';
    ctx.drawImage(card.glow, frame.x, frame.y, frame.w, frame.h, 0, 0, c.width, c.height);
  }
}

async function main(): Promise<void> {
  const data = (await (await fetch(`${BASE}atlas.json`)).json()) as AtlasFile;
  for (const [name, atlas] of Object.entries(data.atlases)) {
    const [albedo, glow] = await Promise.all([loadImage(BASE + atlas.image), loadImage(BASE + atlas.emissive)]);
    const h2 = document.createElement('h2');
    h2.textContent = `${name} — ${Object.keys(atlas.frames).length} assets (${atlas.width}×${atlas.height})`;
    const grid = document.createElement('div');
    grid.className = 'grid';
    root.append(h2, grid);
    for (const [asset, frames] of Object.entries(atlas.frames)) {
      const el = document.createElement('div');
      el.className = 'card';
      const canvas = document.createElement('canvas');
      const label = document.createElement('span');
      label.textContent = frames.length > 1 ? `${asset} ×${frames.length}` : asset;
      el.append(canvas, label);
      grid.append(el);
      cards.push({ canvas, frames, fps: atlas.fps[asset] ?? 0, albedo, glow });
    }
  }
  const tick = (ms: number) => {
    for (const card of cards) draw(card, ms / 1000);
    requestAnimationFrame(tick);
  };
  requestAnimationFrame(tick);
}

main().catch((e: unknown) => {
  root.textContent = `Failed to load assets: ${String(e)}`;
});
```
